\section{Introduzione}

\subsection{Presentazione del dominio scelto}
Il sistema \textbf{EnterpriseCinema} si occupa della gestione e dell'organizzazione delle attività operative di un cinema multisala. 
L'applicazione coordina la vendita dei biglietti per gli spettacoli, la gestione del catalogo dei film e delle relative proiezioni nelle sale fisiche, l'assegnazione e il tracciamento dei turni operativi del personale e la gestione delle recensioni e delle segnalazioni di guasti/anomalie nelle sale.

\subsection{Scopo del progetto}
Lo scopo principale è digitalizzare e automatizzare i processi interni del cinema. Creando un sistema integrato basato sul pattern MVC e persistenza in database relazionale PostgreSQL, il software è in grado di gestire le richieste di acquisto dei clienti e le mansioni operative dei dipendenti in totale autonomia, salvaguardando l'integrità e la coerenza dei dati.

\subsection{Attori principali}
\begin{itemize}
    \item \textbf{Cliente}: Utente esterno registrato che consulta la programmazione, seleziona i posti graficamente in sala, prenota i biglietti ed effettua il pagamento (con coupon di sconto opzionali), oltre a lasciare feedback sui film.
    \item \textbf{Dipendente}: Personale interno con mansioni operative (Manager, Cassiere, Proiezionista, Addetto alle pulizie). Gestisce il catalogo (se Manager), convalida i biglietti e registra segnalazioni di anomalie nelle sale.
    \item \textbf{Sistema Gestionale}: Coordina le operazioni applicando sconti, verificando in tempo reale la capienza e l'occupazione dei posti e garantendo la persistenza delle informazioni nel database relazionale.
\end{itemize}

\subsection{Requisiti funzionali}
Il sistema consente di:
\begin{enumerate}
    \item Registrare e autenticare gli utenti differenziando tra clienti e dipendenti.
    \item Gestire il catalogo dei film (inserimento, modifica ed eliminazione).
    \item Assegnare i film alle sale e pianificare la programmazione delle proiezioni.
    \item Verificare la disponibilità dei posti in sala in tempo reale (12 file da A a L, 15 posti per fila per un totale di 180 posti).
    \item Gestire un carrello di acquisto con applicazione di codici sconto (es. \textit{ENTERPRISE}, \textit{SENIOR}, \textit{JUNIOR}).
    \item Acquistare i biglietti associandoli a posti e proiezioni specifiche tramite simulazione di pagamento.
    \item Generare un titolo d'ingresso stampabile contenente un codice univoco a 8 cifre.
    \item Convalidare i biglietti all'ingresso entro la finestra temporale corretta (da 15 minuti prima dell'inizio dello spettacolo).
    \item Rimborsare un biglietto eliminandolo dal database solo se la proiezione non è ancora iniziata.
    \item Pubblicare recensioni per i film con voto in stelle (da 1 a 5) e commento testuale.
    \item Registrare anomalie e segnalazioni da parte dei dipendenti durante i loro turni nelle sale.
\end{enumerate}

\subsection{Requisiti non funzionali}
\begin{itemize}
    \item \textbf{Integrità Referenziale}: Il database deve imporre vincoli di chiave esterna (Foreign Key).
    \item \textbf{Atomicità delle transazioni}: Le prenotazioni e gli acquisti dei biglietti devono essere atomici.
    \item \textbf{Persistenza}: Memorizzazione dei dati in un database relazionale PostgreSQL.
    \item \textbf{Modularità}: Architettura del software basata sul pattern MVC e disaccoppiamento tramite interfacce DAO.
\end{itemize}

\section{Schema Concettuale (Modello E-R)}

\subsection{Entità Principali}
\begin{itemize}
    \item \textbf{Utente} (Astratta): contiene \textit{email} (PK), \textit{nome}, \textit{cognome}, \textit{password}.
    \item \textbf{Cliente} (Sottoclasse di Utente): aggiunge \textit{dataRegistrazione}.
    \item \textbf{Dipendente} (Sottoclasse di Utente): aggiunge \textit{stipendio}, \textit{ruolo}.
    \item \textbf{Film}: contiene \textit{titolo} (PK), \textit{durata}, \textit{genere}, \textit{classificazioneEta}, \textit{trama}, \textit{percorsoCopertina}, \textit{dataInizioProgrammazione}, \textit{salaAssegnata}.
    \item \textbf{Sala}: contiene \textit{numeroSala} (PK), \textit{capienza} (pari a 180), \textit{tipoSala}.
    \item \textbf{Posto}: identificato da \textit{fila}, \textit{numeroPosto} e \textit{numeroSala}.
    \item \textbf{Proiezione}: identificato da \textit{idProiezione} (PK), contiene \textit{dataOraInizio}, \textit{dataOraFine}, \textit{prezzoBase}.
    \item \textbf{Prenotazione}: identificato da \textit{idPrenotazione} (PK), contiene \textit{dataPrenotazione}, \textit{stato}.
    \item \textbf{Pagamento}: identificato da \textit{idPagamento} (PK), contiene \textit{importo}, \textit{dataPagamento}, \textit{metodoPagamento}, \textit{statoPagamento}.
    \item \textbf{Biglietto}: identificato da \textit{codiceUnivoco} (PK), contiene \textit{prezzoFinale}, \textit{valido}.
\end{itemize}

\subsection{Generalizzazioni}
\begin{itemize}
    \item \textbf{Utente} si specializza in \textbf{Cliente} e \textbf{Dipendente}. La generalizzazione è totale ed esclusiva.
\end{itemize}

\subsection{Associazioni e Relazioni}
\begin{itemize}
    \item \textbf{Cliente - Prenotazione} (1 a $0..*$): un cliente effettua zero o più prenotazioni.
    \item \textbf{Prenotazione - Biglietto} (1 a $1..*$): una prenotazione include uno o più biglietti.
    \item \textbf{Prenotazione - Pagamento} (1 a $0..1$): una prenotazione prevede al massimo un pagamento.
    \item \textbf{Biglietto - Posto} (1 a 1): ciascun biglietto è associato a un posto specifico.
    \item \textbf{Biglietto - Proiezione} (1 a 1): ciascun biglietto si riferisce a una proiezione specifica.
    \item \textbf{Film - Proiezione} (1 a $0..*$): un film ha zero o più proiezioni nel tempo.
    \item \textbf{Sala - Proiezione} (1 a $0..*$): una sala ospita più proiezioni giornaliere.
    \item \textbf{Dipendente - Proiezione (Gestisce)} (Molti-a-Molti): descritta dall'associazione \textit{Gestisce} contenente gli attributi \textit{orarioInizioTurno}, \textit{orarioFineTurno} e \textit{segnalazioni}.
\end{itemize}

\section{Regole di Business}

\subsection{Catalogo delle regole di business}
\begin{enumerate}
    \item \textbf{Unicità Account}: Ogni utente/cliente deve registrarsi con un'email univoca.
    \item \textbf{Prezzo Positivo}: Il prezzo pagato per il biglietto deve essere non negativo.
    \item \textbf{Unicità Assegnazione Posto}: Lo stesso posto (stessa sala, fila, numero) per la stessa data e orario non può essere assegnato a più biglietti contemporaneamente.
    \item \textbf{Limiti Valutazione}: Il voto numerico inserito per una recensione deve essere compreso tra 1 e 5 (inclusi).
    \item \textbf{Rimborso Biglietti}: È vietato eliminare o rimborsare un biglietto se l'orario d'inizio dello spettacolo è già trascorso rispetto all'orario attuale.
    \item \textbf{Finestra di Convalida}: La convalida del biglietto da parte del dipendente all'ingresso del cinema è consentita a partire da 15 minuti prima dell'inizio programmato del film.
    \item \textbf{Capienza Limite della Sala}: Il numero totale di biglietti attivi emessi per un singolo spettacolo non può superare 180 (posti disponibili).
    \item \textbf{Coerenza della Sala}: La sala indicata sul biglietto acquistato da un utente deve necessariamente coincidere con la sala assegnata al film in quel periodo.
\end{enumerate}

\section{Schema Concettuale Ristrutturato}

\subsection{Scelte di ristrutturazione}
\begin{enumerate}
    \item \textbf{Accorpamento della Generalizzazione}: 
    L'entità padre \textit{Utente} e la sottoclasse \textit{Cliente} sono state accorpate direttamente nella tabella fisica \texttt{cliente} nel database PostgreSQL. Poiché i dati relativi ai dipendenti (\textit{Dipendente}) sono gestiti in-memory nel codice Java (RAM), questa scelta evita tabelle vuote e riduce il costo di JOIN a livello di database.
    \item \textbf{Denormalizzazione dei Biglietti}:
    Invece di creare relazioni complesse a livello fisico tra \textit{Biglietto}, \textit{Posto}, \textit{Sala} e \textit{Proiezione}, la tabella relazionale \texttt{biglietto} memorizza direttamente le informazioni in formato testuale denormalizzato (titolo del film, sala, orario della proiezione, fila e numero del posto). Questa scelta rispecchia l'architettura dei DAO del codice Java, ottimizzando i tempi di caricamento ed evitando l'uso di query eccessivamente nidificate.
    \item \textbf{Imposizione dell'unicità del posto}:
    La prevenzione di doppie prenotazioni per il medesimo posto è garantita da un vincolo \texttt{UNIQUE} composto sulle colonne \texttt{(sala, data\_orario\_proiezione, fila, numero\_posto)} all'interno della tabella \texttt{biglietto}.
\end{enumerate}

\subsection{Descrizione dei 4 Trigger}

\paragraph{Trigger 1: check\_posto\_occupato}
\begin{itemize}
    \item \textbf{Evento}: \texttt{BEFORE INSERT OR UPDATE ON biglietto}
    \item \textbf{Condizione}: Se nel database è già registrato un biglietto venduto per la stessa sala, stesso orario di proiezione, stessa fila e stesso numero di posto, avente un codice univoco differente.
    \item \textbf{Azione}: Solleva un'eccezione (\texttt{RAISE EXCEPTION}) annullando l'inserimento/modifica.
\end{itemize}

\paragraph{Trigger 2: check\_rimborso\_biglietto}
\begin{itemize}
    \item \textbf{Evento}: \texttt{BEFORE DELETE ON biglietto}
    \item \textbf{Condizione}: Se la data e l'ora correnti di sistema (\texttt{CURRENT\_TIMESTAMP}) sono maggiori o uguali all'orario previsto per l'inizio del film (\texttt{data\_orario\_proiezione}).
    \item \textbf{Azione}: Blocca l'operazione di cancellazione (rimborso) e solleva un'eccezione.
\end{itemize}

\paragraph{Trigger 3: check\_capienza\_sala}
\begin{itemize}
    \item \textbf{Evento}: \texttt{BEFORE INSERT ON biglietto}
    \item \textbf{Condizione}: Se il conteggio dei biglietti già venduti per la sala e l'orario specificati è maggiore o uguale a 180.
    \item \textbf{Azione}: Impedisce l'inserimento del biglietto sollevando un'eccezione per sala piena.
\end{itemize}

\paragraph{Trigger 4: check\_sala\_biglietto}
\begin{itemize}
    \item \textbf{Evento}: \texttt{BEFORE INSERT OR UPDATE ON biglietto}
    \item \textbf{Condizione}: Se la sala specificata nel biglietto non corrisponde alla colonna \texttt{sala\_assegnata} del rispettivo film nella tabella \texttt{film}.
    \item \textbf{Azione}: Impedisce l'inserimento sollevando un'eccezione di incoerenza logica.
\end{itemize}

\section{Interrogazioni SQL (Query Complesse)}

Di seguito sono riportate 6 interrogazioni SQL complesse (che utilizzano sottoquery annidate, join, aggregazioni, raggruppamenti e clausole \texttt{HAVING}) progettate per scopi analitici:

\begin{enumerate}
    \item \textbf{Query 1: Film di genere 'Thriller' con recensioni superiori alla media globale dei voti}
    \begin{lstlisting}
SELECT DISTINCT f.titolo, f.genere
FROM film f
JOIN recensione r ON f.titolo = r.titolo_film
WHERE f.genere = 'Thriller' AND r.voto > (
    SELECT AVG(voto)
    FROM recensione
);
    \end{lstlisting}

    \item \textbf{Query 2: Film che hanno venduto singolarmente più biglietti rispetto alla media di biglietti venduti per film}
    \begin{lstlisting}
SELECT titolo_film, COUNT(codice_univoco) AS totale_venduti
FROM biglietto
GROUP BY titolo_film
HAVING COUNT(codice_univoco) > (
    SELECT AVG(conteggio_biglietti)
    FROM (
        SELECT COUNT(codice_univoco) AS conteggio_biglietti
        FROM biglietto
        GROUP BY titolo_film
    ) AS subquery
);
    \end{lstlisting}

    \item \textbf{Query 3: Biglietti venduti per spettacoli in sale con segnalazioni guasti attive non risolte}
    \begin{lstlisting}
SELECT DISTINCT b.sala, b.titolo_film, b.data_orario_proiezione
FROM biglietto b
WHERE b.sala IN (
    SELECT s.sala
    FROM segnalazione s
    WHERE s.risolta = FALSE
) AND EXISTS (
    SELECT 1 
    FROM segnalazione s
    WHERE s.sala = b.sala 
      AND s.risolta = FALSE 
      AND s.data_invio <= b.data_orario_proiezione
);
    \end{lstlisting}

    \item \textbf{Query 4: Sale il cui incasso totale supera l'incasso medio calcolato per sala}
    \begin{lstlisting}
SELECT sala, SUM(prezzo_pagato) AS ricavo_sala
FROM biglietto
GROUP BY sala
HAVING SUM(prezzo_pagato) > (
    SELECT AVG(ricavo_medio)
    FROM (
        SELECT SUM(prezzo_pagato) AS ricavo_medio
        FROM biglietto
        GROUP BY sala
    ) AS subquery
);
    \end{lstlisting}

    \item \textbf{Query 5: Film che hanno ricevuto feedback/recensioni ma per cui non e ancora stato venduto alcun biglietto}
    \begin{lstlisting}
SELECT DISTINCT r.titolo_film
FROM recensione r
WHERE r.titolo_film NOT IN (
    SELECT DISTINCT b.titolo_film
    FROM biglietto b
);
    \end{lstlisting}

    \item \textbf{Query 6: Segnalazioni di guasto in sale che programmano film con durata superiore a 2 ore}
    \begin{lstlisting}
SELECT s.id, s.sala, s.messaggio, s.data_invio
FROM segnalazione s
WHERE s.sala IN (
    SELECT f.sala_assegnata
    FROM film f
    WHERE f.durata > '02:00:00'
);
    \end{lstlisting}
\end{enumerate}

\section{Script SQL (DDL)}

\subsection{Creazione del Database e Tabelle}
Lo script DDL seguente definisce lo schema relazionale del database in PostgreSQL:

\begin{lstlisting}
-- Creazione della tabella CLIENTE
CREATE TABLE cliente (
    email character varying(255) NOT NULL PRIMARY KEY,
    nome character varying(100) NOT NULL,
    cognome character varying(100) NOT NULL,
    password character varying(255) NOT NULL,
    data_registrazione date NOT NULL
);

-- Creazione della tabella FILM
CREATE TABLE film (
    titolo character varying(150) NOT NULL PRIMARY KEY,
    durata time without time zone NOT NULL,
    genere character varying(50) NOT NULL,
    classificazione_eta character varying(20) NOT NULL,
    trama text,
    percorso_copertina character varying(500),
    data_inizio_programmazione date,
    sala_assegnata character varying(50) DEFAULT 'Sala 1'::character varying
);

-- Creazione della tabella RECENSIONE e SEQUENCE associata
CREATE SEQUENCE recensione_id_seq
    START WITH 1
    INCREMENT BY 1
    NO MINVALUE
    NO MAXVALUE
    CACHE 1;

CREATE TABLE recensione (
    id integer NOT NULL DEFAULT nextval('recensione_id_seq'::regclass) PRIMARY KEY,
    titolo_film character varying(150),
    autore character varying(100),
    voto integer CHECK (voto BETWEEN 1 AND 5),
    commento text,
    CONSTRAINT fk_recensione_film FOREIGN KEY (titolo_film) 
        REFERENCES film(titolo) ON DELETE CASCADE
);

-- Creazione della tabella SEGNALAZIONE e SEQUENCE associata
CREATE SEQUENCE segnalazione_id_seq
    START WITH 1
    INCREMENT BY 1
    NO MINVALUE
    NO MAXVALUE
    CACHE 1;

CREATE TABLE segnalazione (
    id integer NOT NULL DEFAULT nextval('segnalazione_id_seq'::regclass) PRIMARY KEY,
    mittente_email character varying(255),
    messaggio text,
    sala character varying(50),
    risolta boolean DEFAULT false,
    data_invio timestamp without time zone DEFAULT CURRENT_TIMESTAMP
);

-- Creazione della tabella BIGLIETTO
CREATE TABLE biglietto (
    codice_univoco character varying(8) NOT NULL PRIMARY KEY,
    titolo_film character varying(150),
    sala character varying(50),
    data_orario_proiezione timestamp without time zone,
    fila character(1),
    numero_posto integer,
    prezzo_pagato numeric(5,2) CHECK (prezzo_pagato >= 0),
    stato character varying(20) DEFAULT 'DA CONVALIDARE'::character varying,
    CONSTRAINT unique_posto_proiezione UNIQUE (sala, data_orario_proiezione, fila, numero_posto)
);
\end{lstlisting}

\section{Script Trigger PostgreSQL (PL/pgSQL)}

\subsection{Implementazione Stored Procedure e Trigger}
Di seguito sono implementate le funzioni associate ai trigger ed i relativi trigger su PostgreSQL per l'applicazione delle regole di business:

\begin{lstlisting}
-- Trigger 1: Controllo posto gia venduto
CREATE OR REPLACE FUNCTION check_posto_occupato()
RETURNS TRIGGER AS $$
BEGIN
    IF EXISTS (
        SELECT 1 FROM biglietto
        WHERE sala = NEW.sala
          AND data_orario_proiezione = NEW.data_orario_proiezione
          AND fila = NEW.fila
          AND numero_posto = NEW.numero_posto
          AND codice_univoco <> NEW.codice_univoco
    ) THEN
        RAISE EXCEPTION 'Il posto %, fila % per la proiezione del % nella sala % e gia stato venduto!', 
            NEW.numero_posto, NEW.fila, NEW.data_orario_proiezione, NEW.sala;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_check_posto_occupato
BEFORE INSERT OR UPDATE ON biglietto
FOR EACH ROW
EXECUTE FUNCTION check_posto_occupato();


-- Trigger 2: Impedire il rimborso dopo l'inizio del film
CREATE OR REPLACE FUNCTION check_rimborso_biglietto()
RETURNS TRIGGER AS $$
BEGIN
    IF OLD.data_orario_proiezione <= CURRENT_TIMESTAMP THEN
        RAISE EXCEPTION 'Impossibile rimborsare/eliminare il biglietto %: la proiezione e gia iniziata o terminata.', 
            OLD.codice_univoco;
    END IF;
    RETURN OLD;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_check_rimborso_biglietto
BEFORE DELETE ON biglietto
FOR EACH ROW
EXECUTE FUNCTION check_rimborso_biglietto();


-- Trigger 3: Controllo capienza sala max 180
CREATE OR REPLACE FUNCTION check_capienza_sala()
RETURNS TRIGGER AS $$
DECLARE
    posti_venduti INT;
BEGIN
    SELECT COUNT(*) INTO posti_venduti
    FROM biglietto
    WHERE sala = NEW.sala
      AND data_orario_proiezione = NEW.data_orario_proiezione;
      
    IF posti_venduti >= 180 THEN
        RAISE EXCEPTION 'Errore: La sala % per lo spettacolo del % ha raggiunto la capienza massima (180 posti)!', 
            NEW.sala, NEW.data_orario_proiezione;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_check_capienza_sala
BEFORE INSERT ON biglietto
FOR EACH ROW
EXECUTE FUNCTION check_capienza_sala();


-- Trigger 4: Verifica coerenza sala biglietto con sala film
CREATE OR REPLACE FUNCTION check_sala_biglietto()
RETURNS TRIGGER AS $$
DECLARE
    sala_film VARCHAR(50);
BEGIN
    SELECT sala_assegnata INTO sala_film
    FROM film
    WHERE titolo = NEW.titolo_film;
    
    IF sala_film IS NOT NULL AND NEW.sala <> sala_film THEN
        RAISE EXCEPTION 'Errore: Il film % e programmato in %, non puoi acquistare un biglietto per la %!', 
            NEW.titolo_film, sala_film, NEW.sala;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_check_sala_biglietto
BEFORE INSERT OR UPDATE ON biglietto
FOR EACH ROW
EXECUTE FUNCTION check_sala_biglietto();
\end{lstlisting}
