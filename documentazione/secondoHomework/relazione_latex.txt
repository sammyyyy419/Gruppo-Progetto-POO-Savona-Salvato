\section{Specifica dei Requisiti}

\subsection{Scopo e ambito del sistema}
Il sistema \textbf{EnterpriseCinema} si occupa della gestione delle attività operative di un cinema multisala. L'applicazione coordina la vendita dei biglietti per gli spettacoli, la gestione del catalogo dei film e delle relative proiezioni nelle sale fisiche, l'assegnazione e il tracciamento dei turni operativi del personale e la gestione delle recensioni e delle segnalazioni di guasti/anomalie nelle sale.

\subsection{Attori primari}
\begin{itemize}
    \item \textbf{Cliente}: Utente esterno registrato che consulta la programmazione, seleziona i posti graficamente in sala, prenota i biglietti ed effettua il pagamento (con coupon di sconto opzionali), oltre a lasciare feedback sui film.
    \item \textbf{Dipendente}: Personale interno con mansioni operative (Manager, Cassiere, Proiezionista, Addetto alle pulizie). Gestisce il catalogo (se Manager), convalida i biglietti e registra segnalazioni di anomalie nelle sale.
    \item \textbf{Sistema Gestionale (PostgreSQL + Java Controller)}: Coordina le operazioni applicando sconti, verificando in tempo reale la capienza e l'occupazione dei posti e garantendo la persistenza delle informazioni nel database relazionale.
\end{itemize}

\subsection{Requisiti funzionali (Casi d'Uso)}
Il sistema consente di:
\begin{enumerate}
    \item Registrare e autenticare gli utenti differenziando tra clienti e dipendenti.
    \item Gestire il catalogo dei film (inserimento, modifica ed eliminazione).
    \item Assegnare i film alle sale e pianificare la programmazione delle proiezioni.
    \item Verificare la disponibilità dei posti in sala in tempo reale (12 file da A a L, 15 posti per fila per un totale di 180 posti).
    \item Gestire un carrello di acquisto con applicazione di codici sconto (es. \textit{ENTERPRISE}, \textit{SENIOR}, \textit{JUNIOR}).
    \item Acquistare i biglietti associandoli a posti e proiezioni specifiche tramite simulazione di pagamento.
    \item Generare un titolo d'ingresso stampabile contenente un codice univoco a 8 cifre.
    \item Convalidare i biglietti all'ingresso entro la finestra temporale corretta (da 15 minuti prima dell'inizio dello spettacolo).
    \item Rimborsare un biglietto eliminandolo dal database solo se la proiezione non è ancora iniziata.
    \item Pubblicare recensioni per i film con voto in stelle (da 1 a 5) e commento testuale.
    \item Registrare anomalie e segnalazioni da parte dei dipendenti durante i loro turni nelle sale.
\end{enumerate}

\subsection{Requisiti non funzionali}
\begin{itemize}
    \item \textbf{Integrità Referenziale}: Il database deve imporre vincoli di chiave esterna (Foreign Key).
    \item \textbf{Atomicità delle transazioni}: Le prenotazioni e gli acquisti dei biglietti devono essere atomici.
    \item \textbf{Persistenza}: Memorizzazione dei dati in un database relazionale PostgreSQL.
    \item \textbf{Modularità}: Architettura del software basata sul pattern MVC e disaccoppiamento tramite interfacce DAO.
\end{itemize}

\subsection{Descrizioni testuali dei casi d'uso critici}

\paragraph{UC1: Selezione Posto e Acquisto Biglietto}
\begin{itemize}
    \item \textbf{Attori}: Cliente (primario).
    \item \textbf{Pre-condizioni}: Il cliente è autenticato e ha selezionato uno spettacolo (proiezione).
    \item \textbf{Flusso principale}:
        1. Il cliente visualizza la griglia della sala (A-L, 1-15).
        2. Il cliente seleziona graficamente uno o più posti.
        3. Il sistema verifica la disponibilità (nessuna prenotazione esistente).
        4. Il cliente aggiunge i biglietti al carrello.
        5. Il cliente inserisce un codice sconto (es. \textit{ENTERPRISE}, \textit{SENIOR}).
        6. Il cliente procede al checkout, simulando il pagamento elettronico.
        7. Il database valida i dati tramite il trigger \texttt{check\_posto\_occupato} e il vincolo UNIQUE composto.
        8. Il sistema persiste il biglietto con stato "DA CONVALIDARE" e genera un codice a 8 cifre.
    \item \textbf{Post-condizioni}: I posti sono acquistati e non sono più disponibili per la stessa proiezione.
\end{itemize}

\paragraph{UC2: Convalida Biglietto all'Ingresso}
\begin{itemize}
    \item \textbf{Attori}: Dipendente (Cassiere/Proiezionista/Manager).
    \item \textbf{Pre-condizioni}: Lo spettatore presenta il biglietto cartaceo/digitale con il codice a 8 cifre.
    \item \textbf{Flusso principale}:
        1. Il dipendente inserisce il codice univoco del biglietto nell'applicazione.
        2. Il sistema esegue la ricerca del biglietto nel database.
        3. Il dipendente avvia la validazione tramite il metodo \texttt{validaBiglietto} della classe \texttt{Dipendente}.
        4. Il sistema verifica se l'orario attuale è compreso tra 15 minuti prima dell'inizio dello spettacolo e la fine del film.
        5. Se valido, lo stato del biglietto viene aggiornato in "CONVALIDATO".
    \item \textbf{Post-condizioni}: Il biglietto viene registrato come utilizzato, impedendo doppie convalide.
\end{itemize}

\paragraph{UC3: Richiesta di Rimborso Biglietto}
\begin{itemize}
    \item \textbf{Attori}: Cliente.
    \item \textbf{Pre-condizioni}: Il cliente visualizza lo storico dei suoi biglietti attivi non ancora convalidati.
    \item \textbf{Flusso principale}:
        1. Il cliente seleziona il biglietto che intende rimborsare.
        2. Il cliente clicca sul pulsante "Chiedi Rimborso".
        3. Il sistema invia una richiesta di cancellazione (\texttt{DELETE}) della riga del biglietto nel DB.
        4. Il trigger \texttt{check\_rimborso\_biglietto} intercetta la cancellazione e confronta l'orario dello spettacolo con \texttt{CURRENT\_TIMESTAMP}.
        5. Se l'orario dello spettacolo è nel futuro, il record viene rimosso dal DB.
    \item \textbf{Post-condizioni}: Il biglietto viene eliminato fisicamente e il posto viene liberato.
\end{itemize}

\paragraph{UC4: Invio Segnalazione Sala Guasta}
\begin{itemize}
    \item \textbf{Attori}: Dipendente.
    \item \textbf{Pre-condizioni}: Il dipendente riscontra un problema (es. proiettore rotto, poltrone danneggiate) in una sala durante il turno.
    \item \textbf{Flusso principale}:
        1. Il dipendente accede all'interfaccia delle segnalazioni.
        2. Il dipendente seleziona la sala e scrive un testo descrittivo dell'anomalia.
        3. Il dipendente invia la segnalazione.
        4. Il sistema inserisce la segnalazione nel DB con stato \texttt{risolta = FALSE} e timestamp corrente.
    \item \textbf{Post-condizioni}: La segnalazione è inserita e resa visibile al Manager.
\end{itemize}

\section{Lista delle Funzioni ad Alto Livello}

Le funzioni ad alto livello sono raggruppate nei tre moduli logici che governano il sistema gestionale:

\subsection{Modulo 1: Gestione Utenti ed Autenticazione}
\begin{itemize}
    \item \textbf{\texttt{aggiungiCliente}}:
        \begin{itemize}
            \item \textit{Descrizione}: Registra un nuovo account cliente nel database relazionale.
            \item \textit{Input}: Oggetto \texttt{Cliente} contenente email, nome, cognome, password e data di registrazione.
            \item \textit{Output}: \texttt{void} (inserimento eseguito).
        \end{itemize}
    \item \textbf{\texttt{validaLogin}}:
        \begin{itemize}
            \item \textit{Descrizione}: Verifica la corrispondenza delle credenziali inserite dall'utente per l'autenticazione.
            \item \textit{Input}: \texttt{email} (String), \texttt{password} (String).
            \item \textit{Output}: \texttt{boolean} (esito dell'autenticazione).
        \end{itemize}
    \item \textbf{\texttt{modificaCredenzialiCliente}}:
        \begin{itemize}
            \item \textit{Descrizione}: Permette ad un cliente di aggiornare il proprio indirizzo email di accesso e la password.
            \item \textit{Input}: Oggetto \texttt{Cliente}, \texttt{nuovaEmail} (String), \texttt{nuovaPassword} (String).
            \item \textit{Output}: \texttt{void} (aggiornamento eseguito).
        \end{itemize}
\end{itemize}

\subsection{Modulo 2: Gestione Film e Programmazione}
\begin{itemize}
    \item \textbf{\texttt{aggiungiFilm}}:
        \begin{itemize}
            \item \textit{Descrizione}: Inserisce una nuova pellicola all'interno del catalogo del cinema (riservata ai Manager).
            \item \textit{Input}: Oggetto \texttt{Film} (titolo, durata, genere, classificazione età, trama, copertina, data inizio programmazione, sala assegnata).
            \item \textit{Output}: \texttt{void}.
        \end{itemize}
    \item \textbf{\texttt{eliminaFilm}}:
        \begin{itemize}
            \item \textit{Descrizione}: Rimuove un film dal catalogo cancellandolo fisicamente dal database.
            \item \textit{Input}: Oggetto \texttt{Film}.
            \item \textit{Output}: \texttt{void}.
        \end{itemize}
    \item \textbf{\texttt{getProiezioniPerFilm}}:
        \begin{itemize}
            \item \textit{Descrizione}: Recupera dal database le proiezioni e gli spettacoli attivi pianificati per il film specificato.
            \item \textit{Input}: Oggetto \texttt{Film}.
            \item \textit{Output}: \texttt{ArrayList<Proiezione>}.
        \end{itemize}
\end{itemize}

\subsection{Modulo 3: Gestione Acquisti, Biglietti e Feedback}
\begin{itemize}
    \item \textbf{\texttt{confermaAcquistoCarrello}}:
        \begin{itemize}
            \item \textit{Descrizione}: Esegue la transazione di checkout del carrello, assegna i posti e crea i biglietti.
            \item \textit{Input}: \texttt{metodoPagamento} (String), \texttt{percentualeSconto} (double).
            \item \textit{Output}: \texttt{void} (acquisti registrati e carrello svuotato).
        \end{itemize}
    \item \textbf{\texttt{convalidaBigliettoPerCodice}}:
        \begin{itemize}
            \item \textit{Descrizione}: Convalida all'ingresso lo spettatore tramite inserimento del codice univoco del biglietto.
            \item \textit{Input}: \texttt{codiceUnivoco} (String).
            \item \textit{Output}: Oggetto \texttt{Biglietto} (stato aggiornato a "CONVALIDATO").
        \end{itemize}
    \item \textbf{\texttt{rimborsaSingoloBiglietto}}:
        \begin{itemize}
            \item \textit{Descrizione}: Rimuove dal database un biglietto acquistato liberando il posto in sala.
            \item \textit{Input}: Oggetto \texttt{Biglietto}.
            \item \textit{Output}: \texttt{void}.
        \end{itemize}
    \item \textbf{\texttt{aggiungiRecensioneAFilm}}:
        \begin{itemize}
            \item \textit{Descrizione}: Associa ad un film il feedback (voto e commento) rilasciato da un cliente.
            \item \textit{Input}: \texttt{film} (Film), \texttt{cliente} (Cliente), \texttt{voto} (int), \texttt{commento} (String).
            \item \textit{Output}: \texttt{boolean} (esito inserimento).
        \end{itemize}
\end{itemize}

\section{Schema Concettuale (Modello E-R Iniziale)}

Lo schema concettuale rappresenta il dominio applicativo in formato astratto. In questa fase iniziale, la gerarchia degli utenti e le entità fisiche della sala cinematografica (Sale, Posti e Proiezioni) sono modellate in modo esaustivo per rappresentare fedelmente lo scenario reale del cinema multisala.

\subsection{Entità e Attributi}
\begin{itemize}
    \item \textbf{Utente} (Astratta): contiene \textit{email} (Identificativo PK), \textit{nome}, \textit{cognome}, \textit{password}.
    \item \textbf{Cliente} (Sotto-entità): eredita gli attributi di Utente e aggiunge \textit{data\_registrazione}.
    \item \textbf{Dipendente} (Sotto-entità): eredita gli attributi di Utente e aggiunge \textit{stipendio}, \textit{ruolo}.
    \item \textbf{Film}: contiene \textit{titolo} (Identificativo PK), \textit{durata}, \textit{genere}, \textit{classificazione\_eta}, \textit{trama}, \textit{percorso\_copertina}, \textit{data\_inizio\_programmazione}, \textit{sala\_assegnata}.
    \item \textbf{Sala}: contiene \textit{numero\_sala} (PK), \textit{capienza} (pari a 180), \textit{tipoSala}.
    \item \textbf{Posto}: identificato debolmente da \textit{fila}, \textit{numero\_posto} in associazione a \textit{Sala}.
    \item \textbf{Proiezione}: identificato da \textit{id\_proiezione} (PK), contiene \textit{dataOraInizio}, \textit{prezzoBase}.
    \item \textbf{Prenotazione}: identificato da \textit{id\_prenotazione} (PK), contiene \textit{data\_prenotazione}, \textit{stato}.
    \item \textbf{Pagamento}: identificato da \textit{id\_pagamento} (PK), contiene \textit{importo}, \textit{data\_pagamento}, \textit{metodo\_pagamento}.
    \item \textbf{Biglietto}: identificato da \textit{codice\_univoco} (PK), contiene \textit{prezzo\_pagato}, \textit{stato}.
    \item \textbf{Recensione}: identificato da \textit{id} (PK), contiene \textit{voto}, \textit{commento}, \textit{autore}.
    \item \textbf{Segnalazione}: identificato da \textit{id} (PK), contiene \textit{messaggio}, \textit{sala}, \textit{risolta}, \textit{data\_invio}.
\end{itemize}

\subsection{Associazioni e Cardinalità}
\begin{itemize}
    \item \textbf{Generalizzazione Utente}: \textbf{Utente} si specializza in \textbf{Cliente} e \textbf{Dipendente}. Si tratta di una generalizzazione totale (tutti gli utenti appartengono ad una delle categorie) ed esclusiva (un utente non può essere contemporaneamente cliente e dipendente).
    \item \textbf{Cliente - Prenotazione} (1 a $0..*$): un cliente può effettuare zero o più prenotazioni nel tempo. Ciascuna prenotazione appartiene ad un unico cliente.
    \item \textbf{Prenotazione - Biglietto} (1 a $1..*$): una prenotazione raggruppa uno o più biglietti acquistati nello stesso checkout.
    \item \textbf{Prenotazione - Pagamento} (1 a $0..1$): una prenotazione ha al massimo un pagamento ad essa associato.
    \item \textbf{Biglietto - Posto} (1 a 1): ciascun biglietto è associato ad un unico posto fisico e specifico.
    \item \textbf{Biglietto - Proiezione} (1 a 1): ciascun biglietto si riferisce ad una specifica proiezione.
    \item \textbf{Film - Proiezione} (1 a $0..*$): un film ha più proiezioni pianificate. Ciascuna proiezione si riferisce ad un unico film.
    \item \textbf{Sala - Proiezione} (1 a $0..*$): una sala ospita più proiezioni. Ciascuna proiezione avviene in un'unica sala.
    \item \textbf{Posto - Sala} (molti a 1): molti posti appartengono ad un'unica sala fisica.
\end{itemize}

\section{Regole di Business}

Le regole di business definiscono i vincoli semantici e di dominio che la base di dati e l'applicazione devono garantire a livello di consistenza logica:

\begin{enumerate}
    \item \textbf{Vincolo di unicità dell'account}: Ogni utente (cliente o dipendente) deve potersi registrare esclusivamente con un indirizzo email univoco e non nullo.
    \item \textbf{Prezzo valido}: Il prezzo pagato registrato per l'acquisto di un biglietto o calcolato all'interno del carrello non può in nessun caso assumere valori negativi.
    \item \textbf{Unicità dell'assegnazione del posto}: Lo stesso posto (definito univocamente dalla combinazione di fila e numero di poltrona) nella stessa sala, per lo stesso spettacolo (data e orario di proiezione), non può essere assegnato a più biglietti contemporaneamente (prevenzione dell'overbooking fisico).
    \item \textbf{Limiti semantici di feedback (Voto)}: Il voto numerico espresso da un utente in una recensione ad un film deve risiedere rigorosamente all'interno del range chiuso $[1, 5]$.
    \item \textbf{Finestra temporale di rimborso}: La richiesta di rimborso (cancellazione fisica) di un biglietto da parte di un cliente è consentita esclusivamente se l'orario di inizio programmato dello spettacolo è strettamente successivo al timestamp corrente.
    \item \textbf{Finestra temporale di convalida}: Un dipendente può convalidare un biglietto d'ingresso a partire da 15 minuti prima dell'inizio programmato del film e fino all'orario previsto di fine proiezione.
    \item \textbf{Capienza massima fisica}: Il numero totale di biglietti attivi venduti per una singola proiezione non può eccedere i 180 posti, ovvero la capienza massima stabilita per le sale del cinema.
    \item \textbf{Coerenza della sala cinematografica}: La sala fisica indicata per l'emissione di un biglietto deve coincidere esattamente con la sala in cui è programmata la programmazione del film in quel determinato intervallo.
    \item \textbf{Calcolo automatico dello sconto tessera}: Se il cliente possiede una tessera promozionale specifica (es. Senior, Junior), il prezzo calcolato nel carrello deve essere ridotto della percentuale associata (Senior = 50\% sconto, Junior = 25\% sconto, Enterprise = 10\% sconto).
\end{enumerate}

\section{Schema Concettuale Ristrutturato}

Prima di convertire lo schema E-R iniziale in tabelle relazionali per il DBMS PostgreSQL, si applicano alcune scelte di ristrutturazione per ottimizzare l'I/O e semplificare lo schema:

\subsection{Decisioni di Ristrutturazione}
\begin{enumerate}
    \item \textbf{Accorpamento e parziale eliminazione delle gerarchie}:
        L'entità astratta \textit{Utente} e la sottoclasse \textit{Cliente} sono state fuse direttamente all'interno dell'unica tabella fisica \texttt{cliente}. La sottoclasse \textit{Dipendente} è stata ristrutturata rimuovendola dal database fisico: i dipendenti dello staff (essendo numericamente fissi ed esigui) vengono gestiti interamente in-memory all'interno della classe \texttt{Controller.java} in Java. Questa scelta riduce il sovraccarico di JOIN tra tabelle vuote e semplifica notevolmente lo schema logico del DBMS.
    \item \textbf{Denormalizzazione delle tabelle Sala, Posto e Proiezione}:
        Nel database relazionale non sono state create tabelle fisiche per \textit{Sala}, \textit{Posto} e \textit{Proiezione}. Le informazioni relative a sala, orario dello spettacolo e coordinate della poltrona sono state denormalizzate e inserite direttamente all'interno delle colonne della tabella \texttt{biglietto} (es. colonna \texttt{sala}, \texttt{fila}, \texttt{numero\_posto}, \texttt{data\_orario\_proiezione}). Questa ristrutturazione velocizza enormemente le query di lettura (non necessitano di JOIN su 4 tabelle diverse) a livello di DAO.
    \item \textbf{Accorpamento di Prenotazione e Pagamento}:
        Le entità \textit{Prenotazione} e \textit{Pagamento} sono state accorpate all'interno dell'entità \texttt{biglietto}. Le informazioni sul pagamento (metodo e prezzo) e sullo stato della prenotazione (valido, da convalidare) sono rappresentate dagli attributi del biglietto stesso.
\end{enumerate}

\section{Specifiche Dettagliate delle Operazioni Critiche}

Vengono analizzate nel dettaglio le tre operazioni più sensibili del sistema, illustrando segnatura, logica di business, pseudo-codice e operazioni corrispondenti sulla base di dati:

\subsection{Operazione 1: confermaAcquistoCarrello}
\begin{itemize}
    \item \textbf{Segnatura}: \texttt{public void confermaAcquistoCarrello(String metodoPagamento, double percentualeSconto) throws SalaPienaException}
    \item \textbf{Logica di Business}: Riceve il checkout del carrello da parte dell'utente. Cerca i posti liberi (preferendo quelli vicini/consecutivi) tramite \texttt{trovaPostiVicini()}, crea un oggetto \texttt{Prenotazione} confermato, e acquista i singoli biglietti inserendoli nel DB. Infine svuota il carrello.
    \item \textbf{Operazioni DB}: Esegue query di \texttt{INSERT} per ogni tupla biglietto tramite il DAO.
    \item \textbf{Pseudo-codice}:
\end{itemize}
\begin{lstlisting}[language=Java]
PER OGNI ElementoCarrello elem IN carrello DO
    quantita = elem.getQuantita()
    proiezione = elem.getProiezione()
    postiAssegnati = trovaPostiVicini(proiezione, quantita) // solleva eccezione se piena
    
    nuovaPrenotazione = NEW Prenotazione(LocalDateTime.now(), StatoPrenotazione.CONFERMATO, proiezione, utenteLoggato)
    
    PER i = 0 TO quantita - 1 DO
        posto = postiAssegnati.get(i)
        biglietto = NEW Biglietto(prezzoScontato, posto, proiezione, nuovaPrenotazione)
        
        // Operazione DB tramite DAO
        bigliettoDAO.inserisciBigliettoDB(biglietto)
        listaBiglietti.add(biglietto)
        nuovaPrenotazione.getBiglietti().add(biglietto)
    ROUTINE
ROUTINE
svuotaCarrello()
\end{lstlisting}

\subsection{Operazione 2: validaBiglietto (nel Dipendente)}
\begin{itemize}
    \item \textbf{Segnatura}: \texttt{public boolean validaBiglietto(Biglietto biglietto, Proiezione proiezione)}
    \item \textbf{Logica di Business}: Controlla che il biglietto non sia nullo, non sia già stato convalidato e che l'orario attuale rientri nella finestra consentita (da 15 minuti prima dell'inizio dello spettacolo a fine spettacolo). Se valido, lo stato in Java e DB viene aggiornato.
    \item \textbf{Operazioni DB}: Esegue un \texttt{UPDATE} sul database aggiornando lo stato in 'CONVALIDATO'.
    \item \textbf{Pseudo-codice}:
\end{itemize}
\begin{lstlisting}[language=Java]
IF biglietto == null OR proiezione == null THEN RETURN false
IF biglietto.isValido() THEN RETURN false // Gia convalidato

orarioAttuale = LocalDateTime.now()
orarioInizio = proiezione.getDataOraInizio()
orarioIngressoConsentito = orarioInizio.minusMinutes(15)
orarioFine = orarioInizio.plusMinutes(proiezione.getFilm().getDurataMinuti())

IF orarioAttuale.isAfter(orarioIngressoConsentito) AND orarioAttuale.isBefore(orarioFine) THEN
    biglietto.setValido(true)
    // Operazione DB tramite DAO
    bigliettoDAO.aggiornaStatoBigliettoDB(biglietto.getCodiceUnivoco(), "CONVALIDATO")
    RETURN true
ELSE
    RETURN false
ENDIF
\end{lstlisting}

\subsection{Operazione 3: rimborsaSingoloBiglietto}
\begin{itemize}
    \item \textbf{Segnatura}: \texttt{public void rimborsaSingoloBiglietto(Biglietto bigliettoDaRimborsare) throws Exception}
    \item \textbf{Logica di Business}: Verifica che il biglietto esista e non sia già stato convalidato. Rimuove il biglietto dalla memoria in Java e invia la richiesta di cancellazione al DAO. Se l'orario dello spettacolo è superato, il database rifiuterà l'operazione tramite il trigger.
    \item \textbf{Operazioni DB}: Esegue una query di \texttt{DELETE} sulla tabella biglietto.
    \item \textbf{Pseudo-codice}:
\end{itemize}
\begin{lstlisting}[language=Java]
IF bigliettoDaRimborsare == null THEN THROW Exception("Inesistente")
IF bigliettoDaRimborsare.isValido() THEN THROW Exception("Gia obliterato")

listaBiglietti.remove(bigliettoDaRimborsare)
prenotazione = bigliettoDaRimborsare.getPrenotazione()
IF prenotazione != null THEN
    prenotazione.getBiglietti().remove(bigliettoDaRimborsare)
    IF prenotazione.getBiglietti().isEmpty() THEN
        prenotazione.setStato(StatoPrenotazione.RIMBORSATO)
    ENDIF
ENDIF

// Operazione DB tramite DAO (lancia il trigger BEFORE DELETE)
bigliettoDAO.eliminaBigliettoDB(bigliettoDaRimborsare.getCodiceUnivoco())
\end{lstlisting}

\section{Script dei Trigger in PostgreSQL (PL/pgSQL)}

Vengono definiti i codici per la creazione delle stored procedure e l'associazione dei 4 trigger su PostgreSQL:

\begin{lstlisting}[language=SQL]
-- Trigger 1: Controllo posto gia venduto
CREATE OR REPLACE FUNCTION check_posto_occupato()
RETURNS TRIGGER AS $$
BEGIN
    IF EXISTS (
        SELECT 1 FROM biglietto
        WHERE sala = NEW.sala
          AND data_orario_proiezione = NEW.data_orario_proiezione
          AND fila = NEW.fila
          AND numero_posto = NEW.numero_posto
          AND codice_univoco <> NEW.codice_univoco
    ) THEN
        RAISE EXCEPTION 'Il posto %, fila % per la proiezione del % nella sala % e gia stato venduto!', 
            NEW.numero_posto, NEW.fila, NEW.data_orario_proiezione, NEW.sala;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_check_posto_occupato
BEFORE INSERT OR UPDATE ON biglietto
FOR EACH ROW
EXECUTE FUNCTION check_posto_occupato();


-- Trigger 2: Impedire il rimborso dopo l'inizio del film
CREATE OR REPLACE FUNCTION check_rimborso_biglietto()
RETURNS TRIGGER AS $$
BEGIN
    IF OLD.data_orario_proiezione <= CURRENT_TIMESTAMP THEN
        RAISE EXCEPTION 'Impossibile rimborsare/eliminare il biglietto %: la proiezione e gia iniziata o terminata.', 
            OLD.codice_univoco;
    END IF;
    RETURN OLD;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_check_rimborso_biglietto
BEFORE DELETE ON biglietto
FOR EACH ROW
EXECUTE FUNCTION check_rimborso_biglietto();


-- Trigger 3: Controllo capienza sala max 180
CREATE OR REPLACE FUNCTION check_capienza_sala()
RETURNS TRIGGER AS $$
DECLARE
    posti_venduti INT;
BEGIN
    SELECT COUNT(*) INTO posti_venduti
    FROM biglietto
    WHERE sala = NEW.sala
      AND data_orario_proiezione = NEW.data_orario_proiezione;
      
    IF posti_venduti >= 180 THEN
        RAISE EXCEPTION 'Errore: La sala % per lo spettacolo del % ha raggiunto la capienza massima (180 posti)!', 
            NEW.sala, NEW.data_orario_proiezione;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_check_capienza_sala
BEFORE INSERT ON biglietto
FOR EACH ROW
EXECUTE FUNCTION check_capienza_sala();


-- Trigger 4: Verifica coerenza sala biglietto con sala film
CREATE OR REPLACE FUNCTION check_sala_biglietto()
RETURNS TRIGGER AS $$
DECLARE
    sala_film VARCHAR(50);
BEGIN
    SELECT sala_assegnata INTO sala_film
    FROM film
    WHERE titolo = NEW.titolo_film;
    
    IF sala_film IS NOT NULL AND NEW.sala <> sala_film THEN
        RAISE EXCEPTION 'Errore: Il film % e programmato in %, non puoi acquistare un biglietto per la %!', 
            NEW.titolo_film, sala_film, NEW.sala;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_check_sala_biglietto
BEFORE INSERT OR UPDATE ON biglietto
FOR EACH ROW
EXECUTE FUNCTION check_sala_biglietto();
\end{lstlisting}
